## 3. Node CDP 连接脚本骨架（可直接复制）

> **核心模板**，所有验证脚本从这扩展。注意标注的关键点。

```js
const http = require('http');
const WS = require('C:/Users/Zhangwh/.claude/tmp-cdp/node_modules/ws');
const { execSync } = require('child_process');
const timeout = ms => new Promise(r => setTimeout(r, ms));

function getJson(url){ return new Promise((res,rej)=>{
  http.get(url, r=>{ let d=''; r.on('data',c=>d+=c); r.on('end',()=>res(JSON.parse(d))); }).on('error',rej);
});}

(async () => {
  const PORT = 9225;                    // 避开 TIME_WAIT，递增
  const PROFILE = 'C:/Users/Zhangwh/.claude/tmp-cdp/chrome-profile-test';
  const URL = 'file:///D:/tj/822/考研题库/index.html';
  execSync(`rm -rf "${PROFILE}"`);      // 干净 profile
  const chrome = require('child_process').spawn(
    'C:/Program Files/Google/Chrome/Application/chrome.exe',
    [`--remote-debugging-port=${PORT}`, `--user-data-dir=${PROFILE}`, '--no-first-run', '--disable-gpu', 'about:blank'],
    { stdio: 'ignore' });
  const killMine = () => { try { execSync(`taskkill //F //PID ${chrome.pid} //T 2>/dev/null || true`); } catch(e){} };

  // 等页面 target 就绪（page type、about:blank）
  let page = null;
  for (let i=0;i<40;i++) {
    try { const list = await getJson(`http://127.0.0.1:${PORT}/json`);
          page = list.find(p => p.type==='page' && p.url.startsWith('about:blank')); if (page) break; } catch(e){}
    await timeout(300);
  }
  if (!page) { console.log('NO PAGE'); killMine(); process.exit(1); }

  // ★ 连接页面自己的 ws（不是 browser 级！browser 级带 flatten sessionId 反而可能 "Runtime.evaluate wasn't found"）
  const ws = new WS(page.webSocketDebuggerUrl);
  await new Promise(r => ws.on('open', r));

  let nextId = 1; const pending = {}; const exceptions = [];
  ws.on('message', m => {
    m = JSON.parse(m);   // ★★★ ws 收消息是 Buffer，必须 JSON.parse(m)！
    if (m.id && pending[m.id]) { pending[m.id].resolve(m); delete pending[m.id]; }
    if (m.method === 'Runtime.exceptionThrown')
      exceptions.push((m.params.exceptionDetails.exception && m.params.exceptionDetails.exception.description || '').slice(0,200));
  });
  function send(method, params) { return new Promise((resolve,reject) => {
    const id = nextId++; pending[id] = { resolve, reject };
    ws.send(JSON.stringify({ id, method, params }));
  });}

  await send('Page.enable'); await send('Runtime.enable');
  await send('Page.navigate', { url: URL });   // ★ file:// 用 Page.navigate，别用 CLI 传 URL
  await timeout(3000);

  const ev = async (expr) => {
    const r = await send('Runtime.evaluate', { expression: expr, returnByValue: true });
    if (r.result && r.result.exceptionDetails) return 'EXC:' + (r.result.exceptionDetails.exception && r.result.exceptionDetails.exception.description || '').slice(0,200);
    return r.result && r.result.result ? r.result.result.value : '__NOVAL__';
  };

  // ===== 在这里写断言 =====
  console.log('typeof statuses =', await ev('typeof statuses'));

  try { ws.close(); } catch(e){}
  killMine();  // ★ 只能按 PID 杀自己启动的 Chrome
  process.exit(0);
})().catch(e => { console.log('FATAL', e.message); process.exit(1); });
```

**★ 三条命门（曾踩过的坑）：**
1. **`JSON.parse(m)`**：ws 消息是 Buffer，不 parse 则 `m.id` 永远 undefined，所有 `Runtime.evaluate` 表现为超时（而"1+1 却正常"——因为全局监听不走 id 匹配，极具迷惑性）。
2. **连页面 target 的 ws**：不要用 browser 级 `webSocketDebuggerUrl` + `Target.attachToTarget` + `flatten:true`，那样 Runtime 命令可能报 `'Runtime.evaluate' wasn't found`。直接从 `http://127.0.0.1:PORT/json` 拿 `page.webSocketDebuggerUrl` 连接。
3. **只按 PID 清理**：`taskkill //F //IM chrome.exe //T` 会杀掉用户正开的浏览器，被权限分类器拒绝。用 `taskkill //F //PID ${chrome.pid} //T`。

## 4. 五种验证手段

### 4.1 DOM / JS 状态断言（主力）
```js
await ev('switchSubject("822")');               // 切科目
await ev('currentChapterId="ch5"; current=0; loadStatuses(); loadQBad(); loadSBad(); loadNotes(); switchTo(0);');
await ev('getChapter().labels[current]');        // 当前题标签
await ev('typeof statuses');                     // 全局变量
```
注意：初始化在脚本底部执行，先 `switchTo` 走真实渲染；直接改 `current` 后需手动调 load/render 函数补全状态。

### 4.2 异常捕获
脚本已挂 `Runtime.exceptionThrown` → `exceptions[]`，结尾打印。**必须 0 异常才算过。**

### 4.3 真实键盘事件（`Input.dispatchKeyEvent`）
```js
await send('Input.dispatchKeyEvent', { type:'keyDown', key:'d', code:'KeyD', windowsVirtualKeyCode:68 });
await send('Input.dispatchKeyEvent', { type:'keyUp',   key:'d', code:'KeyD', windowsVirtualKeyCode:68 });
```
验证快捷键（A/D/W/S 切题、Z/X/C 打标等）走这条。

### 4.4 滚轮事件 — ★ 特殊
**`Input.dispatchMouseEvent` 的 `type:'mouseWheel'` 在无头/后台窗口会挂起不返回**，会拖死整个脚本（表现为超时）。改用**页面内主动派发**：
```js
await ev(`(()=>{ document.dispatchEvent(new WheelEvent('wheel',{deltaX:100,deltaY:0,bubbles:true,cancelable:true})); return 'ok'; })()`);
```
注意：`window.dispatchEvent(...)` **不冒泡到 document 监听，不会触发**，必须 `document.dispatchEvent`。

### 4.5 数据完整性扫描（labels vs 磁盘）
用 Node 脚本 `eval` 章节字面量 / 正则抽取 labels 数组，与磁盘 `考研题库/…/` 下 PNG 做集合对比。这是核对图片全不全会、命名规范是否统一的可靠手段（改命名时用它）。

## 5. 已知验证环境陷阱

| 陷阱 | 现象 | 解决 |
|---|---|---|
| ws 消息是 Buffer | 全命令超时 | 回调里 `JSON.parse(m)` |
| Input mouseWheel 挂起 | 脚本卡死 | `document.dispatchEvent(new WheelEvent(...))` |
| window.dispatchEvent 不触发 document 监听 | 测试假失败 | 用 `document.dispatchEvent` |
| file:// 直接作 CLI 参数 | 打开介绍页 | `Page.navigate` 加载 |
| CSS transition 不推进（无头/最小化） | 灯箱 `.show` 后 `visibility` 卡 hidden，`elementFromPoint` 穿透 | 测试先 `el.style.transition='none'` |
| `getEventListeners` 在页面上下文不可用 | 拿不到监听 | 用 `DOMDebugger.getEventListeners(objectId)`（DevTools 命令面） |
| 端口 TIME_WAIT | 连上但命令挂起 | 递增换端口 |
| `taskkill //IM chrome.exe` | 杀用户浏览器被拒 | `taskkill //PID <自启进程>` |
| 图片加载验证 | `fetch(file://)` 挂起 | `new Image()` DOM 探针（`img.complete`/`naturalWidth`） |

   ```bash
   node -e "const fs=require('fs');const src=fs.readFileSync('index.html','utf8');const re=/<script[^>]*>([\s\S]*?)<\/script>/g;let m,n=0,ok=true;while((m=re.exec(src))!==null){if(m[1].trim().length===0)continue;n++;try{new Function(m[1]);}catch(e){ok=false;console.log('ERR:',e.message);}}console.log('blocks:',n,ok?'ALL OK':'FAIL');"
   ```
2. 写 CDP 脚本（第 3 节骨架），覆盖本次改动的**正反两面**：
   - 正向：新行为生效（如滚轮右推→下一题）
   - 反向：防护生效（如纵向滚轮不切题、灯箱打开不切题）

3. 断言 + `exceptions` 0 异常。
4. 跑完 `killMine()`，确认 `netstat -ano | grep :PORT` 无 LISTENING 残留。
5. 清理 `/tmp` 临时脚本。
